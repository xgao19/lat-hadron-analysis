# Example DA N-state ratio-fit input file
# Edit the values below, then run:
#   lqcd-analysis da-nstate-fit templates/input_files/da/da_nstate_example_annotated.txt
#
# First line format:
#   title_pattern Ns Nt lattice_spacing_fm

# Data settings
l64c64a076_m140_fit_pz* 64 64 0.076

# Set this to true to force a bT=bz=0 decay-constant-style check.
# In this mode the workflow uses only the real part, scans nearby fit windows,
# and prints the decay constant in GeV.
decay_constant_check false

# Set this to true to couple each DA bootstrap sample to the matching
# two-point bootstrap sample row. When omitted, the workflow keeps using the
# summary-table reference and prints one warning at startup.
two_point_fit_sample_coupled false

# Integer momentum labels to analyze.
pzlist 0

# Gamma/operator labels passed into the dataset-path template and used to select
# the operator-dependent DA physics behavior.
# Supported labels in the first version:
#   T5 -> gamma_t gamma_5
#   Z5 -> gamma_z gamma_5
gmlist T5

# Eta labels passed into the dataset-path template.
etalist eta0

# Average over these Tdir entries after reading the HDF5 datasets.
Tdirlist plus minus

# Transverse separation choices.
# Use either bTlist or bTrange.
bTlist 0

# Longitudinal separation choices.
# If bz != 0, the workflow automatically combines +bz and -bz.
bzlist 0

# HDF5 file path template for the DA inputs.
# Use {gm} to keep the operator label variable.
qda_h5 /path/to/qDA_..._O{gm}.h5

# Dataset template inside the HDF5 file.
# Supported placeholders:
#   {gm} {eta} {pz} {Tdir} {bT} {bz}
dataset_path_template SP/{gm}/PX0PY0PZ{pz}/{Tdir}/{eta}/bT{bT}/bz{bz}

# Raw time window to keep before the DA fit.
# Optional. If omitted, the backend defaults to [0, Nt//2 - 1].
tsrange 0 20

# Two-point correlator input

# Root directory containing the two-point n-state fit outputs reused by this workflow.
# Point this at the root directory that contains the existing two-point n-state
# fit outputs. The backend will look under
#   <two_point_fit_root>/<title>/tables/
# and select the matching `*_tmax<tmax>_fits.txt` file.
two_point_fit_root examples/outputs/da_two_point_fit_ref

# Per-momentum tmin/tmax choice for reusing the two-point fit table directly.
# The notebook version accepts this as a dict and materializes it to a tiny
# three-column mapping file before running the workflow.
two_point_fit_window_by_pz templates/input_files/da/da_two_point_fit_window_by_pz_example.txt

# Two-point correlator CSV used to build the denominator part of the ratio.
c2pt examples/l64c64a076_m140/data/c2pt_csv/c2pt_5_5_k0_pz*_real.csv

# Time-folding mode for the two-point denominator correlator.
# Choices:
#   none / false
#   periodic / true
#   antiperiodic
fold_t periodic

# DA fit settings

# Only the ratio workflow is supported.
fit_target ratio

# Which component(s) to fit:
#   real
#   imag
#   both
fit_component both

# Allowed state choices in this first version:
#   nstates 1
#   nstates 2
#   nstates 1 2
nstates 1 2

# Bootstrap controls.
binsize 1
bootstrap_samples 32
bootstrap_size 32
seed 2026

# Recommended default usage: set one trusted fit window per momentum.
# Each row in the fit-window table should contain:
#   pz tmin tmax
# or, for gm-specific control:
#   gm pz tmin tmax
# In the notebook workflow_config, the preferred equivalent is:
#   "fit_window": {5: [6, 12], 6: [6, 12], 7: [5, 12]}
fit_window /path/to/da_fit_windows.txt

# Plot generation is reserved for later extension.
# Keep false in this first version.
plot false

# Output directory.
# If omitted, outputs go to a `results_da_fit/` directory next to this input file.
results_dir examples/outputs/da_fit_example

# Notes:
# - The first implementation fits the real and imaginary parts separately.
# - decay_constant_check forces a real-part-only check at bT = bz = 0 and scans
#   tmin in [tmin - 2, tmin + 2] and tmax in [tmax - 4, tmax] around the
#   requested fit window.
# - A_i and E_i are loaded from the selected row of the two-point fit table as
#   fixed central values by default.
# - Set two_point_fit_sample_coupled true to use the matching two-point
#   bootstrap-sample A_i/E_i for each DA bootstrap sample.
# - Bootstrap sample fits start from the same plain zero initial guess for each
#   sample; there is no separate mean-fit warm start in this template.
# - The reported chi2, chi2_dof, and pvalue are summarized from the successful
#   bootstrap fits.
# - T5 keeps the original sign-pattern preprocessing.
# - Z5 multiplies the correlator by -i before folding and uses an extra Pz/E_i factor in the numerator model.
# - Outputs are grouped by bT. For a fixed (title, gm, eta, bT, component, nstates),
#   one summary / fit / samples / curve file is written and each file contains all bz entries.
# - Grouped summary files contain one `begin_bz ...` / `end_bz ...` block per bz.
# - Grouped summary blocks record:
#     two_point_fit_table_resolved
#     two_point_fit_tmax_source
#     two_point_fit_tmax
#   where the tmax value now comes from the explicit two-point fit-window config.
# - Grouped fit tables include two_point_fit_tmax plus the fit results.
