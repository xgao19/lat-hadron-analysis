# Example 2pt N-state fit input file
# Edit the values below, then run:
#   python scripts/two_point/fit_2pt_nstate.py 2pt-nstate-fit templates/input_files/two_point/nstate_fit_example_realdata_annotated.txt
#
# First line format:
#   title_pattern Ns Nt lattice_spacing_fm
l64c64a076_m140_fit_k0_pz* 64 64 0.076

# Path or wildcard pattern for correlator CSV files.
c2pt examples/data/l64c64a076_m140/comb_c2pt_csv/c2pt_5_5_k0_pz*_real.csv

# Momentum indices to analyze.
pzlist 0

# Time-folding mode before fitting.
# Choices:
#   none / false
#   periodic / true
#   antiperiodic
fold_t none

# Raw correlator time window [t_start, t_end].
# Optional. If omitted, the backend defaults to [0, Nt//2 - 1].
tsrange 0 24

# Fit model choices:
#   normal
#   symmetric
#   antisymmetric
model normal

# Fit error model:
#   uncorrelated = diagonal fit using per-time-slice bootstrap standard deviations
#   correlated   = full time-time covariance fit using one shared covariance matrix
#                  built from the full bootstrap ensemble and reused for all fits
#                  if a correlated fit fails, the code falls back to a diagonal
#                  fit using only the covariance diagonal for that sample/window
fit_mode uncorrelated

# Allowed fit-state choices are subsets of: 1 2 3
# Examples:
#   nstates 1
#   nstates 1 2
#   nstates 1 2 3
nstates 1 2

# Optional: pz=0 ground-state energy in lattice units.
# If provided, the 1-state plateau search uses the target
#   E_target(pz) = sqrt(E0_pz0^2 + (2*pi*pz/Ns)^2)
# to pre-filter plateau candidates before the usual ranking.
# This affects only the 1-state plateau selection, not the fit model itself.
# pz0_ground_energy 0.42

# Fit upper bound. Use an integer or the word auto.
tmax 12

# Bootstrap and run controls.
# Use auto for bootstrap_samples / bootstrap_size to let the backend choose.
binsize 1
bootstrap_samples 16
bootstrap_size 16
seed 2026

# Soft-prior strength for higher-state fits.
# The prior enters the least-squares residual as
#   sqrt(lambda_prior) * (E - E_prior) / sigma_prior
# Default is 1.0.
# 1-state has no lower-state prior.
# 2-state gets a prior on E0 from the 1-state plateau result.
# 3-state gets priors on E0 and E1 from the 2-state result.
lambda_prior 1.0

# Whether to generate plots automatically.
# Choices: true / false
plot true

# Output note:
#   Fit tables include `fallback_uncorrelated_successes`, the number of bootstrap
#   samples in that tmin window that succeeded only after falling back from the
#   correlated fit to a diagonal fit.
#   The summary also reports this count for the representative window.

# Output directory.
# If omitted, outputs go to a `results_nstate_fit/` directory next to this input file.
results_dir examples/outputs/nstate_fit_realdata
