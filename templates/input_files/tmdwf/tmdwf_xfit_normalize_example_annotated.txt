# Example TMDWF x-fit normalization input file
# Edit the values below, then run:
#   lqcd-analysis tmdwf-xfit-normalize templates/input_files/tmdwf/tmdwf_xfit_normalize_example_annotated.txt
#
# This workflow normalizes q0(x) outputs from tmdwf-x-nstate-fit using old
# bare matrix-element m0(bz) outputs from tmdwf-nstate-fit. It does not read
# old Fourier outputs. Internally it applies the requested m0 normalization
# mode in bz space, Fourier transforms both old bare and old normalized m0(bz),
# and applies their x-space ratio to the new q0(x) samples.

title_pattern l64c64a076_m140_tmdwf_pz*

# New x-space fit outputs.
input_root examples/outputs/tmdwf_x_nstate_fit_example

# Old bare matrix-element outputs from tmdwf-nstate-fit.
bare_matrix_root examples/outputs/tmdwf_fit_notebook

ns 64
lattice_spacing_fm 0.076
pzlist 0
gmlist T5
etalist eta0

# Choose either:
#   bTlist ...
# or:
#   bTrange bTmin bTmax
bTlist 0

# bz grid from the old bare matrix-element outputs used to build the
# normalization factor.
bzlist 0 2 4 6 8 10 12

# Select the x-fit output family to normalize.
component real
nstates 1

# Normalization mode:
#   mode1 -> m0(pz, bT, bz) / m0(pz, bT=0, bz=0)
#   mode2 -> m0(pz, bT, bz) / m0(pz=0, bT, bz=0)
#   mode3 -> [m0(pz, bT, bz) / m0(pz, bT=0, bz=0)] / [m0(pz=0, bT, bz=0) / m0(pz=0, bT=0, bz=0)]
normalization_mode mode1

# Fourier settings used only to construct the old m0 normalization factor on
# the same x grid as the new x-fit samples.
zstep_fm 0.01
interpolation_kind cubic

# Output settings.
# Writes:
#   <results_dir>/<title>/tables/*_<mode>_*_xfit.txt
#   <results_dir>/<title>/samples/*_<mode>_*_xfit_samples.txt
results_dir examples/outputs/tmdwf_xfit_normalize_example
