# Example TMDWF x-space N-state fit input file
# Edit the values below, then run:
#   lqcd-analysis tmdwf-x-nstate-fit templates/input_files/tmdwf/tmdwf_x_nstate_fit_example_annotated.txt
#
# This workflow consumes q(x,t) samples from tmdwf-ratio-fourier-t and fits
# the t dependence independently at each x. It is split out so fit_window can
# be retuned without rerunning the ratio Fourier-per-t step.

# Data settings.
title_pattern l64c64a076_m140_tmdwf_pz*
input_root examples/outputs/tmdwf_ratio_fourier_t_example
ns 64
nt 64
lattice_spacing_fm 0.076
pzlist 0
gmlist T5
etalist eta0

# Choose either:
#   bTlist ...
# or:
#   bTrange bTmin bTmax
bTlist 0

# Select the q(x,t) component and state counts to fit.
component real
nstates 1 2

# Fit settings.
# fit_window is a plain text table with rows such as:
#   0 4 10
# or operator-specific rows such as:
#   T5 0 4 10
fit_window /path/to/tmdwf_x_fit_windows.txt

# Two-point references provide amplitudes and energies for the time model.
two_point_fit_root /path/to/two_point_fit_outputs
two_point_fit_window_by_pz /path/to/tmdwf_two_point_fit_window_by_pz.txt

# true couples each q(x,t) bootstrap sample to matching two-point bootstrap
# amplitudes/energies; false uses the selected two-point summary row.
two_point_fit_sample_coupled false

# Output settings.
# Writes:
#   <results_dir>/<title>/tables/*_xfit.txt
#   <results_dir>/<title>/samples/*_xfit_samples.txt
results_dir examples/outputs/tmdwf_x_nstate_fit_example
