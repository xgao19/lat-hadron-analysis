# Standalone downstream TMDWF Collins-Soper kernel extraction.
# This workflow reads already-generated TMDWF Fourier outputs and
# extracts the type-2 CS kernel without rerunning the TMDWF fit or Fourier step.

# Data settings

# Output title / analysis tag.
title_pattern l64c64a076_m140_fit_pz*

# Root directory that holds the existing Fourier outputs.
# The workflow automatically resolves the usual Fourier table/sample filenames
# from title_pattern, gmlist, etalist, bTlist, component, nstates, and normalization_mode.
input_root /path/to/tmdwf_fourier_outputs

# Ensemble metadata.
ns 64
lattice_spacing_fm 0.076
gmlist T5
etalist eta0
component real
nstates 1
normalization_mode raw
mu 2.0

# CS-kernel extraction settings

# Legacy-compatible extraction controls.
scheme CG
extraction_type type2
pair_mode all
reference_p1 auto
kernel_labels LO NLO NLL NNLO NNLL

# Batch ranges.
bTrange 0 4
pzrange 2 6

# Fourier-output selectors:
# - gmlist / etalist pick which operator/insertion channels to read
# - component / nstates pick which Fourier output family to read
# - normalization_mode tells the workflow whether to read raw or normalized Fourier outputs
#   and is recorded explicitly in the CS-kernel outputs

# Pair construction mode:
# - all: use the shared reference_p1 if provided, otherwise the first requested
#        pz in pzlist, and compare it against the remaining requested momenta
#        in one output group for the full pz window
# - adjacent: build one output group per neighboring pair, e.g. 5-6, 6-7, 7-8
# - fixed_p1: use the shared reference_p1 if provided, otherwise the first
#             requested pz in pzlist, and build one output group per later P2
# - reference_p1: optional fixed shared P1 momentum. When omitted or set to
#   auto, the workflow keeps using the first requested pz in pzlist.
# - fixed_p1 filenames are tagged with fixedp1_... so they do not collide with
#   the band-style all-mode outputs.

# Legacy default x window for the constant-in-P2 fit.
# Only x values inside this window are used when fitting the estimator
# built from the pair_mode-selected P1/P2 combinations.
x_window 0.2 0.8

# Output settings

# Optional plot output.
# When true, also write one quick-look PDF band plot for each
# (kernel_label, bT, pair-group) output group.
# In pair_mode adjacent or fixed_p1, this also writes one pairwise breakdown plot
# showing data log-ratio, matching correction, and total estimator.
plot true

# Optional explicit output root.
# The workflow writes:
# - <results_dir>/<title_pattern with pz->multiPz>/*_summary.txt
# - <results_dir>/<title_pattern with pz->multiPz>/tables/*_refpz<pzmin-pzmax>_type2_band.txt
# - <results_dir>/<title_pattern with pz->multiPz>/samples/*_refpz<pzmin-pzmax>_type2_samples.txt
# - <results_dir>/<title_pattern with pz->multiPz>/diagnostics/*_refpz<pzmin-pzmax>_type2_diagnostics.txt
# - <results_dir>/<title_pattern with pz->multiPz>/plots/*_band.pdf   (when plot=true and pair_mode=all)
# - <results_dir>/<title_pattern with pz->multiPz>/plots/*_pairwise_breakdown.pdf   (when plot=true and pair_mode=adjacent or fixed_p1)
#   fixed_p1 breakdown plots include a fixedp1_refpz... tag in the filename
results_dir results_tmdwf_cs_kernel
