# Example TMDWF N-state ratio-fit input file
# Edit the values below, then run:
#   lqcd-analysis tmdwf-nstate-fit templates/input_files/tmdwf/tmdwf_nstate_example_annotated.txt
#
# First line format:
#   title_pattern Ns Nt lattice_spacing_fm
l64c64a076_m140_tmdwf_pz* 64 64 0.076

# Only the ratio workflow is supported.
fit_target ratio

# Which component(s) to fit:
#   real
#   imag
#   both
fit_component both

# Allowed state choices in this first version:
#   nstates 1
#   nstates 2
#   nstates 1 2
nstates 1 2

# Integer momentum labels to analyze.
pzlist 0

# Gamma/operator labels passed into the dataset-path template and used to select
# the operator-dependent TMDWF physics behavior.
# Supported labels in the first version:
#   T5 -> gamma_t gamma_5
#   Z5 -> gamma_z gamma_5
gmlist T5

# Eta labels passed into the dataset-path template.
etalist eta0

# Average over these Tdir entries after reading the HDF5 datasets.
Tdirlist plus minus

# Transverse separation choices.
# Use either bTlist or bTrange.
bTlist 0

# Longitudinal separation choices.
# If bz != 0, the workflow automatically combines +bz and -bz.
bzlist 0

# Bootstrap controls.
binsize 1
bootstrap_samples 32
bootstrap_size 32
seed 2026

# Recommended default usage: set one trusted fit window per momentum.
# Each row in the fit-window table should contain:
#   pz tmin tmax
# or, for gm-specific control:
#   gm pz tmin tmax
# In the notebook workflow_config, the preferred equivalent is:
#   "fit_window": {5: [6, 12], 6: [6, 12], 7: [5, 12]}
fit_window /path/to/tmdwf_fit_windows.txt

# Advanced / experimental automatic-window controls begin here.
# These are useful for exploratory studies, but are still early-stage and not
# the primary recommended default.
# Optional: when automatic shared-window search is enabled, search a small local
# tmax range around the retained fit-window upper edge. For example, radius 1
# scans:
#   fit_window_tmax - 1, fit_window_tmax, fit_window_tmax + 1
tmax_scan_radius 1

# Optional: when true, treat the retained fit_window as a bounded search region
# and perform automatic shared-window selection inside it.
auto_search_shared_window_from_fit_window false

# Required when auto_search_shared_window_from_fit_window is true.
# The automatic reference-window search keeps the current reference-value /
# decay-constant philosophy and ranks candidate windows by overlap with this
# target, window size, later times, and local m0 stability.
decay_constant 0.11 0.03

# Shared-window scans still enforce the existing hard fit-dof floor of 4.
min_fit_dof 4

# HDF5 file path or wildcard pattern.
# The pz wildcard can be written using * if helpful.
qtmdwf_h5 /path/to/qtmdwf_pz*.h5

# Dataset template inside the HDF5 file.
# Supported placeholders:
#   {gm} {eta} {pz} {Tdir} {bT} {bz}
dataset_path_template {gm}/{eta}/pz{pz}/{Tdir}/bT{bT}/bz{bz}

# Two-point plateau table providing the fixed A_i and E_i inputs.
# This should point to a plateau table produced by the existing two-point workflow.
# You may use:
#   *  for the pz substitution
#   #  for numeric tmax wildcard matching in filenames like `_tmax12_plateau.txt`
# Example:
#   ..._normal_1state_tmax#_plateau.txt
two_point_plateau_table examples/outputs/nstate_fit_realdata/l64c64a076_m140_fit_k0_pz*/tables/l64c64a076_m140_fit_k0_pz*_normal_1state_tmax#_plateau.txt

# Two-point correlator CSV used to build the denominator part of the ratio.
c2pt examples/data/l64c64a076_m140/comb_c2pt_csv/c2pt_5_5_k0_pz*_real.csv

# Time-folding mode for the two-point denominator correlator.
# Choices:
#   none / false
#   periodic / true
#   antiperiodic
fold_t periodic

# Raw time window to keep before the TMDWF fit.
# Optional. If omitted, the backend defaults to [0, Nt//2 - 1].
tsrange 0 20

# Plot generation is reserved for later extension.
# Keep false in this first version.
plot false

# Output directory.
# If omitted, outputs go to a `results_tmdwf_fit/` directory next to this input file.
results_dir examples/outputs/tmdwf_fit_example

# Notes:
# - The first implementation fits the real and imaginary parts separately.
# - A_i and E_i are loaded from the two-point plateau table as fixed central values.
# - The code does not yet couple to two-point bootstrap-sample A_i/E_i.
# - T5 keeps the original sign-pattern preprocessing.
# - Z5 multiplies the correlator by -i before folding and uses an extra Pz/E_i factor in the numerator model.
# - Outputs are grouped by bT. For a fixed (title, gm, eta, bT, component, nstates),
#   one summary / fit / samples / curve file is written and each file contains all bz entries.
# - Grouped summary files contain one `begin_bz ...` / `end_bz ...` block per bz.
# - Grouped ratio and summary outputs also record `fit_window_source`, so you can
#   distinguish manual overrides from automatic reference-based selection.
# - When automatic shared-window search is enabled, the reference summary now
#   scans a bounded `(tmin, tmax)` neighborhood inside the retained fit-window
#   region rather than using the full fit-window directly.
# - Summary blocks also record:
#     two_point_plateau_table_resolved
#     two_point_tmax_source
#     two_point_tmax_inferred
# - Grouped fit tables include metadata columns such as:
#     shared_window_flag
#     reference_eta
#     reference_bT
#     reference_bz
#     plateau_tmax_used
